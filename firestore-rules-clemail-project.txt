rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow ALL authenticated users to READ folder permissions
    match /folderPermissions/{folder} {
      // Allow any authenticated user to read folder permissions
      // This enables manually verified users to check access
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
    }
    
    // Users collection - allow admin domains to read all, users to read/write their own
    match /users/{userId} {
      // Admin domains can read ALL user documents
      allow read: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
      
      // Regular users can only read their own document
      // IMPORTANT: Allow unverified users to read their own doc to check manual verification
      allow read: if request.auth != null && 
        request.auth.uid == userId;
      
      // FIXED: Allow initial document creation for ALL authenticated users
      // This allows unverified users to create their initial user document during registration
      allow create: if request.auth != null && 
        request.auth.uid == userId && 
        request.resource.data.email == request.auth.token.email;
      
      // Updates and other writes require verification OR admin privileges
      allow update: if request.auth != null && 
        (
          // User updating their own document (must be verified)
          (request.auth.uid == userId && request.auth.token.email_verified == true) ||
          // Admin updating any document
          (request.auth.token.email_verified == true &&
           (request.auth.token.email.matches('.*@healthluminate.com') || 
            request.auth.token.email.matches('.*@careluminate.com')))
        );
      
      // Delete requires admin privileges only
      allow delete: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
    }
    
    // NEW: 1099 Tax Data Collection - HIGHLY SECURE - ONLY TAYLOR DAVIS ACCESS
    match /tax1099Data/{userId} {
      // Users can only access their own tax data
      allow read, write: if request.auth != null && 
        request.auth.uid == userId &&
        request.auth.token.email_verified == true;
      
      // ONLY taylordavis@careluminate.com can read all tax data for compliance/processing
      allow read: if request.auth != null && 
        request.auth.token.email_verified == true && 
        request.auth.token.email == 'taylordavis@careluminate.com';
      
      // ONLY taylordavis@careluminate.com can delete tax data (for compliance reasons)
      allow delete: if request.auth != null && 
        request.auth.token.email_verified == true && 
        request.auth.token.email == 'taylordavis@careluminate.com';
    }
    
    // Admin impersonations collection (for tracking admin actions)
    match /adminImpersonations/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
    }
    
    // Project Requests collection - for Executive Retirement tracking system
    match /projectRequests/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }
    
    // Project Edit Logs collection - for tracking changes to projects
    match /projectEditLogs/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }
    
    // Executive Retirement Solutions collections
    match /ers/proposals/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }
    
    match /ers/advisors/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }
    
    match /ers/clients/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }
    
    match /ers/agencies/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com') ||
         request.auth.token.email.matches('.*@executiveretirementplans.com') ||
         request.auth.token.email.matches('.*@outlook.com'));
    }

    // Team Members collection - for outbound calling team
    match /teamMembers/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
      
      // Allow team members to read their own record
      allow read: if request.auth != null && 
        request.auth.token.email == resource.data.email;
    }

    // Campaign Call Tracking collection - for tracking team member calls
    match /campaign_call_tracking/{document} {
      allow read, write: if request.auth != null && 
        request.auth.token.email_verified == true && 
        (request.auth.token.email.matches('.*@healthluminate.com') || 
         request.auth.token.email.matches('.*@careluminate.com'));
      
      // Allow team members to read/write their own call records
      allow read, write: if request.auth != null && 
        request.auth.token.email == resource.data.userEmail;
    }
    
    // Allow all authenticated users to read and write to their own data in other collections
    match /{path=**} {
      allow read, write: if request.auth != null;
    }
  }
}
